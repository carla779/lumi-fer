OPTION 1: Paste this into your Arc browser console
(Right-click on the FilmStim page > Inspect > Console tab)

Copy everything between the --- lines:

Object.entries(clips).forEach(([num, name], i) => {
  setTimeout(() => {
    const a = document.createElement('a');
    a.href = `https://sites.uclouvain.be/ipsp/FilmStim/En/${num}.mp4`;
    a.download = `${name}.mp4`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    console.log(`Downloading ${name} (${i+1}/${Object.keys(clips).length})`);
  }, i * 3000);
});
